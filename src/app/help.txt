import { Ionicons } from "@expo/vector-icons";
import * as SQLite from "expo-sqlite";
import { StatusBar } from "expo-status-bar";
import React, { useCallback, useEffect, useRef, useState } from "react";
import {
  ActivityIndicator,
  Alert,
  KeyboardAvoidingView,
  Modal,
  Platform,
  Pressable,
  ScrollView,
  StyleSheet,
  Switch,
  Text,
  TextInput,
  View,
} from "react-native";
import { SafeAreaView, useSafeAreaInsets } from "react-native-safe-area-context";

type Subject = { id: number; name: string; color: string };
type Task = {
  id: number;
  title: string;
  subject_id: number | null;
  subject_name?: string | null;
  completed: number;
  priority: "Low" | "Medium" | "High";
  due_at: string | null;
};
type Session = {
  id: number;
  subject_id: number | null;
  subject_name?: string | null;
  duration_seconds: number;
  started_at: string;
  date_key: string;
};
type Tab = "Home" | "Timer" | "Todo" | "Subjects" | "Settings";
type ModalType = "task" | "subject" | null;

const C = {
  background: "#F6F7FB",
  card: "#FFFFFF",
  text: "#172033",
  muted: "#8991A5",
  primary: "#635BFF",
  purpleLight: "#EEEDFF",
  green: "#18A878",
  greenLight: "#E3F8EF",
  border: "#ECEEF4",
};
const SUBJECT_COLORS = ["#635BFF", "#18A878", "#FF9D42", "#E45B9A", "#3198DB", "#9270D8"];
const pad = (n: number) => String(n).padStart(2, "0");
const dateKey = (date = new Date()) => `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}`;
function formatTime(seconds: number) {
  const safe = Math.max(0, Math.floor(seconds));
  const h = Math.floor(safe / 3600);
  const m = Math.floor((safe % 3600) / 60);
  const s = safe % 60;
  return h ? `${pad(h)}:${pad(m)}:${pad(s)}` : `${pad(m)}:${pad(s)}`;
}
function formatHours(seconds: number) {
  if (seconds < 3600) return `${Math.round(seconds / 60)} min`;
  const hours = seconds / 3600;
  return `${hours.toFixed(hours < 10 ? 1 : 0)}h`;
}
function startOfWeek(date: Date) {
  const d = new Date(date);
  d.setDate(d.getDate() - ((d.getDay() + 6) % 7));
  d.setHours(0, 0, 0, 0);
  return d;
}

export default function TaskoraScreen() {
  const insets = useSafeAreaInsets();
  const dbRef = useRef<SQLite.SQLiteDatabase | null>(null);
  const timerStartRef = useRef<number | null>(null);
  const baseElapsedRef = useRef(0);
  const sessionStartedAtRef = useRef<string | null>(null);
  const savedSessionRef = useRef(false);

  const [ready, setReady] = useState(false);
  const [tab, setTab] = useState<Tab>("Home");
  const [subjects, setSubjects] = useState<Subject[]>([]);
  const [tasks, setTasks] = useState<Task[]>([]);
  const [sessions, setSessions] = useState<Session[]>([]);
  const [selectedSubject, setSelectedSubject] = useState<number | null>(null);
  const [durationMinutes, setDurationMinutes] = useState(25);
  const [customMinutes, setCustomMinutes] = useState("45");
  const [elapsed, setElapsed] = useState(0);
  const [running, setRunning] = useState(false);
  const [paused, setPaused] = useState(false);
  const [timerSubject, setTimerSubject] = useState<number | null>(null);
  const [modal, setModal] = useState<ModalType>(null);
  const [taskTitle, setTaskTitle] = useState("");
  const [taskPriority, setTaskPriority] = useState<"Low" | "Medium" | "High">("Medium");
  const [taskSubject, setTaskSubject] = useState<number | null>(null);
  const [taskTime, setTaskTime] = useState("");
  const [subjectName, setSubjectName] = useState("");
  const [subjectColor, setSubjectColor] = useState(SUBJECT_COLORS[0]);
  const [darkMode, setDarkMode] = useState(false);
  const [weekOffset, setWeekOffset] = useState(0);

  const theme = darkMode
    ? { ...C, background: "#111522", card: "#1C2232", text: "#F4F5FC", muted: "#A0A8BC", border: "#30374A", purpleLight: "#302D53", greenLight: "#173B32" }
    : C;
  const bg = theme.background;
  const card = theme.card;
  const text = theme.text;
  const muted = theme.muted;

  const loadData = useCallback(async () => {
    const db = dbRef.current;
    if (!db) return;
    const [loadedSubjects, loadedTasks, loadedSessions] = await Promise.all([
      db.getAllAsync<Subject>("SELECT id, name, color FROM subjects ORDER BY id DESC"),
      db.getAllAsync<Task>(`SELECT tasks.*, subjects.name AS subject_name FROM tasks LEFT JOIN subjects ON tasks.subject_id = subjects.id ORDER BY tasks.completed ASC, tasks.id DESC`),
      db.getAllAsync<Session>(`SELECT study_sessions.*, subjects.name AS subject_name FROM study_sessions LEFT JOIN subjects ON study_sessions.subject_id = subjects.id ORDER BY study_sessions.id DESC`),
    ]);
    setSubjects(loadedSubjects);
    setTasks(loadedTasks);
    setSessions(loadedSessions);
    setSelectedSubject((current) => current !== null && loadedSubjects.some((s) => s.id === current) ? current : loadedSubjects[0]?.id ?? null);
  }, []);

  useEffect(() => {
    let mounted = true;
    (async () => {
      try {
        const db = await SQLite.openDatabaseAsync("taskora.db");
        dbRef.current = db;
        await db.execAsync(`
          PRAGMA journal_mode = WAL;
          CREATE TABLE IF NOT EXISTS subjects (id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT NOT NULL UNIQUE, color TEXT NOT NULL);
          CREATE TABLE IF NOT EXISTS tasks (id INTEGER PRIMARY KEY AUTOINCREMENT, title TEXT NOT NULL, subject_id INTEGER, completed INTEGER NOT NULL DEFAULT 0, priority TEXT NOT NULL DEFAULT 'Medium', reminder_id TEXT, due_at TEXT, created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP);
          CREATE TABLE IF NOT EXISTS study_sessions (id INTEGER PRIMARY KEY AUTOINCREMENT, subject_id INTEGER, duration_seconds INTEGER NOT NULL, started_at TEXT NOT NULL, date_key TEXT NOT NULL);
        `);
        const count = await db.getFirstAsync<{ count: number }>("SELECT COUNT(*) AS count FROM subjects");
        if (!count || count.count === 0) {
          for (const [name, color] of [["Physics", "#635BFF"], ["Chemistry", "#18A878"], ["Higher Math", "#FF9D42"], ["Biology", "#E45B9A"]]) {
            await db.runAsync("INSERT OR IGNORE INTO subjects (name, color) VALUES (?, ?)", name, color);
          }
        }
        if (mounted) {
          await loadData();
          setReady(true);
        }
      } catch (error) {
        console.error("Taskora database initialization failed:", error);
        if (mounted) Alert.alert("Database error", "Taskora could not load your local data. Please restart the app and try again.");
      }
    })();
    return () => { mounted = false; };
  }, [loadData]);

  async function saveSession(seconds: number) {
    const db = dbRef.current;
    if (!db || seconds < 1) return;
    try {
      const startedAt = sessionStartedAtRef.current ?? new Date().toISOString();
      const startedDate = new Date(startedAt);
      await db.runAsync(
        "INSERT INTO study_sessions (subject_id, duration_seconds, started_at, date_key) VALUES (?, ?, ?, ?)",
        timerSubject, Math.floor(seconds), startedAt, dateKey(startedDate)
      );
      await loadData();
    } catch (error) {
      console.error("Could not save study session:", error);
      Alert.alert("Save failed", "Your study session could not be saved.");
    }
  }

  useEffect(() => {
    if (!running || timerStartRef.current === null) return;
    const startedAt = timerStartRef.current;
    const baseElapsed = baseElapsedRef.current;
    const targetSeconds = durationMinutes * 60;
    const updateTimer = () => {
      const total = Math.min(targetSeconds, baseElapsed + Math.floor((Date.now() - startedAt) / 1000));
      setElapsed(total);
      if (total >= targetSeconds) {
        timerStartRef.current = null;
        baseElapsedRef.current = total;
        setRunning(false);
        setPaused(false);
        if (!savedSessionRef.current) {
          savedSessionRef.current = true;
          void saveSession(total);
          Alert.alert("Session complete!", "Great work! Your focus session has been saved.");
        }
      }
    };
    updateTimer();
    const interval = setInterval(updateTimer, 500);
    return () => clearInterval(interval);
  }, [running, durationMinutes]);

  function startTimer() {
    if (subjects.length === 0) {
      Alert.alert("No subjects", "Add a study subject before starting a timer.");
      return;
    }
    if (!paused && elapsed === 0) {
      savedSessionRef.current = false;
      baseElapsedRef.current = 0;
      sessionStartedAtRef.current = new Date().toISOString();
      setTimerSubject(selectedSubject);
    } else {
      baseElapsedRef.current = elapsed;
    }
    timerStartRef.current = Date.now();
    setRunning(true);
    setPaused(false);
    setTab("Timer");
  }

  function pauseTimer() {
    if (timerStartRef.current !== null) {
      const total = Math.min(durationMinutes * 60, baseElapsedRef.current + Math.floor((Date.now() - timerStartRef.current) / 1000));
      baseElapsedRef.current = total;
      setElapsed(total);
    }
    timerStartRef.current = null;
    setRunning(false);
    setPaused(true);
  }

  async function finishTimer() {
    let total = elapsed;
    if (running && timerStartRef.current !== null) total = Math.min(durationMinutes * 60, baseElapsedRef.current + Math.floor((Date.now() - timerStartRef.current) / 1000));
    timerStartRef.current = null;
    baseElapsedRef.current = 0;
    setRunning(false);
    setPaused(false);
    setElapsed(0);
    if (total > 0 && !savedSessionRef.current) {
      savedSessionRef.current = true;
      await saveSession(total);
    }
    sessionStartedAtRef.current = null;
  }

  function resetTimer() {
    timerStartRef.current = null;
    baseElapsedRef.current = 0;
    sessionStartedAtRef.current = null;
    setRunning(false);
    setPaused(false);
    setElapsed(0);
    savedSessionRef.current = false;
  }

  function openTaskModal() {
    setTaskTitle(""); setTaskSubject(selectedSubject); setTaskTime(""); setTaskPriority("Medium"); setModal("task");
  }

  async function addTask() {
    const title = taskTitle.trim();
    const db = dbRef.current;
    if (!title || !db) { Alert.alert("Task title", "Enter a title for your task."); return; }
    let dueAt: Date | null = null;
    if (taskTime.trim()) {
      const match = taskTime.trim().match(/^([01]?\d|2[0-3]):([0-5]\d)$/);
      if (!match) { Alert.alert("Invalid time", "Use 24-hour HH:MM format, for example 18:30."); return; }
      dueAt = new Date(); dueAt.setHours(Number(match[1]), Number(match[2]), 0, 0);
      if (dueAt.getTime() <= Date.now()) { Alert.alert("Invalid time", "Enter a time later today."); return; }
    }
    try {
      await db.runAsync("INSERT INTO tasks (title, subject_id, completed, priority, reminder_id, due_at) VALUES (?, ?, 0, ?, NULL, ?)", title, taskSubject, taskPriority, dueAt?.toISOString() ?? null);
      setTaskTitle(""); setTaskTime(""); setTaskPriority("Medium"); setModal(null);
      await loadData();
    } catch (error) {
      console.error("Could not create task:", error);
      Alert.alert("Could not create task", "Please try again.");
    }
  }

  async function toggleTask(task: Task) {
    const db = dbRef.current; if (!db) return;
    try { await db.runAsync("UPDATE tasks SET completed = ? WHERE id = ?", task.completed ? 0 : 1, task.id); await loadData(); }
    catch { Alert.alert("Update failed", "The task could not be updated."); }
  }

  async function deleteTask(task: Task) {
    const db = dbRef.current; if (!db) return;
    try { await db.runAsync("DELETE FROM tasks WHERE id = ?", task.id); await loadData(); }
    catch { Alert.alert("Delete failed", "The task could not be deleted."); }
  }

  async function addSubject() {
    const name = subjectName.trim(); const db = dbRef.current;
    if (!name || !db) { Alert.alert("Subject name", "Enter a subject name."); return; }
    try {
      await db.runAsync("INSERT INTO subjects (name, color) VALUES (?, ?)", name, subjectColor);
      setSubjectName(""); setModal(null); await loadData();
    } catch { Alert.alert("Subject already exists", "A subject with this name already exists."); }
  }

  function deleteSubject(subject: Subject) {
    const db = dbRef.current; if (!db) return;
    Alert.alert("Delete subject?", `Delete ${subject.name}? Existing tasks and sessions will remain, but will no longer be linked to this subject.`, [
      { text: "Cancel", style: "cancel" },
      { text: "Delete", style: "destructive", onPress: async () => {
        await db.runAsync("UPDATE tasks SET subject_id = NULL WHERE subject_id = ?", subject.id);
        await db.runAsync("UPDATE study_sessions SET subject_id = NULL WHERE subject_id = ?", subject.id);
        await db.runAsync("DELETE FROM subjects WHERE id = ?", subject.id);
        if (selectedSubject === subject.id) setSelectedSubject(null);
        await loadData();
      } },
    ]);
  }

  const todaySessions = sessions.filter((s) => s.date_key === dateKey());
  const todaySeconds = todaySessions.reduce((sum, s) => sum + s.duration_seconds, 0);
  const openTasks = tasks.filter((t) => !t.completed);
  const completedTasks = tasks.filter((t) => t.completed).length;
  const weekStart = startOfWeek(new Date());
  weekStart.setDate(weekStart.getDate() + weekOffset * 7);
  const weeklyData = Array.from({ length: 7 }, (_, index) => {
    const day = new Date(weekStart); day.setDate(day.getDate() + index); const key = dateKey(day);
    return { key, label: day.toLocaleDateString("en-US", { weekday: "short" }), seconds: sessions.filter((s) => s.date_key === key).reduce((sum, s) => sum + s.duration_seconds, 0), isToday: key === dateKey() };
  });
  const weeklyTotal = weeklyData.reduce((sum, d) => sum + d.seconds, 0);
  const maxBar = Math.max(...weeklyData.map((d) => d.seconds), 3600);
  const subjectStats = subjects.map((subject) => ({ ...subject, seconds: sessions.filter((s) => s.subject_id === subject.id).reduce((sum, s) => sum + s.duration_seconds, 0) }));
  const timerProgress = Math.min(1, elapsed / Math.max(1, durationMinutes * 60));

  function SectionTitle({ title, action, onAction }: { title: string; action?: string; onAction?: () => void }) {
    return <View style={styles.sectionTitle}><Text style={[styles.sectionHeading, { color: text }]}>{title}</Text>{action ? <Pressable onPress={onAction}><Text style={styles.actionText}>{action}</Text></Pressable> : null}</View>;
  }
  function Card({ children, style }: { children: React.ReactNode; style?: object }) {
    return <View style={[styles.card, { backgroundColor: card, borderColor: theme.border }, style]}>{children}</View>;
  }
  function EmptyState({ title, subtitle, icon }: { title: string; subtitle: string; icon: React.ComponentProps<typeof Ionicons>["name"] }) {
    return <View style={styles.emptyState}><View style={styles.emptyIcon}><Ionicons name={icon} size={25} color={C.primary} /></View><Text style={[styles.emptyTitle, { color: text }]}>{title}</Text><Text style={[styles.emptySubtitle, { color: muted }]}>{subtitle}</Text></View>;
  }
  function renderTask(task: Task) {
    const subject = subjects.find((s) => s.id === task.subject_id);
    return <View key={task.id} style={[styles.taskRow, { borderBottomColor: theme.border }]}>
      <Pressable onPress={() => toggleTask(task)} style={[styles.checkCircle, !!task.completed && styles.checkedCircle]}>{task.completed ? <Ionicons name="checkmark" size={15} color="#fff" /> : null}</Pressable>
      <View style={styles.taskTextWrap}><Text style={[styles.taskTitle, { color: text }, !!task.completed && styles.completedText]}>{task.title}</Text><View style={styles.taskMeta}><View style={[styles.miniDot, { backgroundColor: subject?.color ?? muted }]} /><Text style={[styles.smallText, { color: muted }]}>{subject?.name ?? "General"}</Text>{task.due_at ? <Text style={[styles.smallText, { color: muted }]}> · {new Date(task.due_at).toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" })}</Text> : null}</View></View>
      <View style={[styles.priorityBadge, { backgroundColor: task.priority === "High" ? "#FDE9E9" : task.priority === "Low" ? "#E5F7EF" : "#FFF1DF" }]}><Text style={{ color: task.priority === "High" ? "#D94A4A" : task.priority === "Low" ? "#14845E" : "#B97925", fontSize: 10, fontWeight: "700" }}>{task.priority}</Text></View>
      <Pressable onPress={() => Alert.alert("Delete task?", task.title, [{ text: "Cancel", style: "cancel" }, { text: "Delete", style: "destructive", onPress: () => deleteTask(task) }])} style={styles.deleteButton}><Ionicons name="trash-outline" size={17} color={muted} /></Pressable>
    </View>;
  }

  function renderHome() {
    return <>
      <View style={styles.greetingRow}><View><Text style={[styles.eyebrow, { color: muted }]}>YOUR STUDY SPACE</Text><Text style={[styles.pageTitle, { color: text }]}>Let's focus! 👋</Text><Text style={[styles.subtitle, { color: muted }]}>Small steps. Big results.</Text></View><View style={styles.avatar}><Ionicons name="school" size={24} color={C.primary} /></View></View>
      <Card style={styles.heroCard}><View style={styles.heroTop}><View style={{ flex: 1 }}><Text style={styles.heroEyebrow}>TODAY'S STUDY TIME</Text><Text style={styles.heroNumber}>{formatHours(todaySeconds)}</Text><Text style={styles.heroSub}>Based on your completed timer sessions</Text></View><View style={styles.heroIcon}><Ionicons name="time-outline" size={27} color="#fff" /></View></View><View style={styles.heroDivider} /><View style={styles.heroBottom}><View><Text style={styles.heroStatLabel}>Tasks remaining</Text><Text style={styles.heroStatValue}>{openTasks.length}</Text></View><View><Text style={styles.heroStatLabel}>Tasks completed</Text><Text style={styles.heroStatValue}>{completedTasks}</Text></View><Pressable style={styles.heroButton} onPress={() => setTab("Timer")}><Ionicons name="play" size={15} color={C.primary} /><Text style={styles.heroButtonText}>Study now</Text></Pressable></View></Card>
      <SectionTitle title="Quick overview" action="View planner" onAction={() => setTab("Todo")} /><View style={styles.statsRow}><Card style={styles.statCard}><View style={[styles.statIcon, { backgroundColor: "#EEEDFF" }]}><Ionicons name="timer-outline" size={20} color={C.primary} /></View><Text style={[styles.statValue, { color: text }]}>{todaySessions.length}</Text><Text style={[styles.statLabel, { color: muted }]}>Study sessions</Text></Card><Card style={styles.statCard}><View style={[styles.statIcon, { backgroundColor: "#E3F8EF" }]}><Ionicons name="checkmark-done" size={20} color={C.green} /></View><Text style={[styles.statValue, { color: text }]}>{completedTasks}</Text><Text style={[styles.statLabel, { color: muted }]}>Completed tasks</Text></Card></View>
      <SectionTitle title="Study activity" action="See timer" onAction={() => setTab("Timer")} /><Card><View style={styles.chartHeader}><View><Text style={[styles.chartTotal, { color: text }]}>{formatHours(weeklyTotal)}</Text><Text style={[styles.smallText, { color: muted }]}>Total study time this week</Text></View><View style={styles.weekControls}><Pressable onPress={() => setWeekOffset((v) => v - 1)} style={styles.weekArrow}><Ionicons name="chevron-back" size={16} color={text} /></Pressable><Pressable onPress={() => setWeekOffset((v) => Math.min(0, v + 1))} disabled={weekOffset === 0} style={[styles.weekArrow, weekOffset === 0 && { opacity: 0.3 }]}><Ionicons name="chevron-forward" size={16} color={text} /></Pressable></View></View><View style={styles.chart}>{weeklyData.map((day) => <View key={day.key} style={styles.chartColumn}><Text style={[styles.chartValue, { color: day.isToday ? C.primary : muted }]}>{day.seconds ? day.seconds >= 3600 ? `${(day.seconds / 3600).toFixed(1)}h` : `${Math.round(day.seconds / 60)}m` : "—"}</Text><View style={styles.barTrack}><View style={[styles.chartBar, { height: Math.max(day.seconds > 0 ? 5 : 3, (day.seconds / maxBar) * 112), backgroundColor: day.isToday ? C.primary : "#C7C3FF" }]} /></View><Text style={[styles.chartLabel, { color: day.isToday ? C.primary : muted, fontWeight: day.isToday ? "800" : "500" }]}>{day.label}</Text></View>)}</View><Text style={[styles.chartFootnote, { color: muted }]}>The graph updates when you save a study session.</Text></Card>
      <SectionTitle title="Upcoming tasks" action="+ Add task" onAction={openTaskModal} /><Card>{openTasks.length ? openTasks.slice(0, 4).map(renderTask) : <EmptyState title="You're all caught up!" subtitle="Add a task to plan your next study session." icon="checkbox-outline" />}</Card>
      <SectionTitle title="Your subjects" action="Manage" onAction={() => setTab("Subjects")} /><View style={styles.subjectGrid}>{subjectStats.slice(0, 4).map((s) => <Pressable key={s.id} onPress={() => { setSelectedSubject(s.id); setTab("Timer"); }} style={[styles.subjectTile, { backgroundColor: card, borderColor: theme.border }]}><View style={[styles.subjectColorBar, { backgroundColor: s.color }]} /><Text style={[styles.subjectTileName, { color: text }]}>{s.name}</Text><Text style={[styles.smallText, { color: muted }]}>{formatHours(s.seconds)} studied</Text></Pressable>)}</View>
    </>;
  }

  function renderTimer() {
    const remaining = Math.max(0, durationMinutes * 60 - elapsed);
    return <><Text style={[styles.eyebrow, { color: muted }]}>STAY FOCUSED</Text><Text style={[styles.pageTitle, { color: text }]}>Study timer</Text><Text style={[styles.subtitle, { color: muted }]}>Focus on one thing at a time.</Text>
      <Card style={styles.timerCard}><View style={styles.timerBadge}><Ionicons name="headset-outline" size={16} color={C.primary} /><Text style={styles.timerBadgeText}>{running ? "SESSION IN PROGRESS" : "FOCUS SESSION"}</Text></View><View style={styles.timerCircleOuter}><View style={[styles.timerProgress, { borderColor: C.primary, opacity: timerProgress > 0 ? 1 : 0.22 }]} /><View style={styles.timerCircleInner}><Text style={[styles.timerTime, { color: text }]}>{formatTime(remaining)}</Text><Text style={[styles.timerStatus, { color: muted }]}>{running ? "Keep going!" : paused ? "Paused" : "Ready to focus"}</Text></View></View>
      <Text style={[styles.timerLabel, { color: muted }]}>STUDY SUBJECT</Text><ScrollView horizontal showsHorizontalScrollIndicator={false} contentContainerStyle={styles.chipScroll}>{subjects.map((s) => <Pressable key={s.id} onPress={() => { if (running || elapsed > 0) { Alert.alert("Timer active", "Finish or reset the current session before changing subjects."); return; } setSelectedSubject(s.id); }} style={[styles.subjectChip, { backgroundColor: selectedSubject === s.id ? s.color : theme.background, borderColor: selectedSubject === s.id ? s.color : theme.border }]}><Text style={{ color: selectedSubject === s.id ? "#fff" : text, fontSize: 12, fontWeight: "700" }}>{s.name}</Text></Pressable>)}</ScrollView>
      <Text style={[styles.timerLabel, { color: muted }]}>SESSION LENGTH</Text><View style={styles.durationRow}>{[25, 45, 60].map((minutes) => <Pressable key={minutes} disabled={running || elapsed > 0} onPress={() => { setDurationMinutes(minutes); setCustomMinutes(String(minutes)); }} style={[styles.durationChip, { backgroundColor: durationMinutes === minutes ? C.primary : theme.background, borderColor: durationMinutes === minutes ? C.primary : theme.border, opacity: running || elapsed > 0 ? 0.5 : 1 }]}><Text style={{ color: durationMinutes === minutes ? "#fff" : text, fontWeight: "800" }}>{minutes} min</Text></Pressable>)}</View>
      <View style={styles.customDurationRow}><TextInput value={customMinutes} onChangeText={setCustomMinutes} editable={!running && elapsed === 0} keyboardType="number-pad" placeholder="Custom minutes" placeholderTextColor={muted} style={[styles.customDurationInput, { color: text, backgroundColor: theme.background, borderColor: theme.border }]} /><Pressable disabled={running || elapsed > 0} onPress={() => { const value = Number(customMinutes); if (!Number.isInteger(value) || value < 1 || value > 240) { Alert.alert("Invalid duration", "Choose a duration between 1 and 240 minutes."); return; } setDurationMinutes(value); }} style={[styles.applyButton, (running || elapsed > 0) && { opacity: 0.5 }]}><Text style={styles.applyButtonText}>Apply</Text></Pressable></View>
      <View style={styles.timerActions}>{!running ? <Pressable onPress={startTimer} style={styles.startButton}><Ionicons name="play" size={20} color="#fff" /><Text style={styles.startButtonText}>{paused ?
